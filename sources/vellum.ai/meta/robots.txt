# Link-preview / unfurling bots.
#
# These crawlers do not build a search index — they fetch exactly the one URL
# a person shared in order to render its Open Graph card. They must therefore
# be allowed to reach the app shell (/account/signup, /account/login,
# /assistant/...): those are the URLs people actually paste into Slack,
# iMessage, WhatsApp, LinkedIn, and X. A `Disallow` here makes the link unfurl
# as a bare URL no matter how good the page's OG tags are.
#
# Keeping them *out* of the search index is a separate concern, handled two
# other ways that are unaffected by this group: the `User-agent: *` rules below
# still disallow those paths, and the app shell itself serves
# `<meta name="robots" content="noindex, nofollow">`.
#
# /admin/ stays disallowed everywhere — internal dashboards should never be
# fetched by a third-party service, previews included.

User-agent: facebookexternalhit
User-agent: LinkedInBot
User-agent: Slackbot
User-agent: Slackbot-LinkExpanding
User-agent: Discordbot
User-agent: WhatsApp
User-agent: TelegramBot
Allow: /
Allow: /api/og
Disallow: /admin/

Allow: /
Allow: /api/og
Disallow: /account/
Disallow: /assistant/
Disallow: /admin/
